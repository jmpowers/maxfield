print_anova_table <- function(df, mod) {
  df %>% select(c("trait", "term", "p.value")) %>% 
    mutate(p.value=p.value %>% map_chr(~ format(.x,scientific=-2, digits=2 , nsmall=2)) %>% 
             str_replace("e-0(\\d)"," x 10^-\\1^") %>% str_replace("e-(\\d+)"," x 10^-\\1^")) %>%
    pivot_wider(names_from=term, values_from=p.value) %>% 
    mutate(singular = map_lgl(mod, isSingular),
           R2m = map_dbl(mod, ~ MuMIn::r.squaredGLMM(.x)[[1,"R2m"]])) %>% 
    kable
}

plot_diag <- function(models, trait) {
  diag.plots <- sjPlot::plot_model(models[[trait]], type="diag")
  traitname <- traitnames[[trait]]
  if(is.null(diag.plots[[2]]$round)) {
    panel <- textGrob(traitname, gp=gpar(fontsize=16)) 
  } else {
    panel <- diag.plots[[2]]$round + ggtitle("Random effect: rounds", subtitle=traitname)
  }
  grid.arrange(grobs=list(diag.plots[[4]], panel,
                     diag.plots[[2]]$plotid + ggtitle("Random effect: subplots"), 
                     diag.plots[[2]]$plot   + ggtitle("Random effect: plots"), 
                     diag.plots[[1]], diag.plots[[3]]))
}
