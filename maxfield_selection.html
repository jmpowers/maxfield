    lm(relfitness ~ trait.z * year, 
       data=mnps.plantyr %>% mutate(trait.z = mnps.plantyr[[trait]] / alltraits.sd[[trait]]) %>% 
         rename("fitness" = fitnesstrait) %>% 
         group_by(year) %>% mutate(relfitness = fitness/mean(fitness, na.rm=T)))}), # mean fitness in each yr
    coef = map(model, tidy),
    test = map(model, ~tidy(car::Anova(.x), type=3)),
    emm = map(model, ~summary(emtrends(.x, ~1, var="trait.z"))))

Selection gradients from PCA

Following method of Chong et al. 2018 for reconstructing selection estimates from estimates of selection on PCs: Beta = E * A.

  • PCA can’t handle missing values so they are removed, but this biases the data and reduces sample size - multiple imputation?
calculate_beta <- function(E, slopes) {
  data.frame(B = E %*% slopes$A,
             SE = sqrt(E ^ 2 %*% slopes$SE ^ 2)) %>% 
    mutate(p.value = pt(abs(B)/SE, df=1, lower.tail=F)) %>% #calculate t statistic from difference in means and SE, then look up one-tailed p with 1 df in the upper tail
    rownames_to_column("trait")
}
plot_beta <- function(beta) {
  beta %>% mutate(trait = fct_reorder(trait, B)) %>% 
    ggplot(aes(y=trait, x = B, xmin=B-SE, xmax=B+SE)) + 
    geom_vline(xintercept=0)+ geom_pointrange() + 
    labs(x = "Selection gradient (beta) estimated from gradients on PCs", y = "Trait")
}

mnps.plantyr.std <- mnps.plantyr %>% select(all_of(morphtraits),seeds_est, any_of(names(treatments))) %>% 
  mutate(across(all_of(morphtraits), ~ (.x-mean(.x, na.rm=T))/sd(.x, na.rm=T))) %>% 
  group_by(year) %>% mutate(seeds_rel = seeds_est/mean(seeds_est, na.rm=T), .keep="unused") %>% ungroup %>% 
  drop_na(all_of(morphtraits))

mnps.PCA.slopes <- tidy(mnps.PCA.regression) %>% filter(term != "(Intercept)") %>% rename(A=estimate, SE=std.error)
